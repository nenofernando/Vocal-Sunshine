cmake_minimum_required(VERSION 3.22)
project(NFVSInstaller LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
if(NOT NFVS_PAYLOAD_DLL OR NOT NFVS_VERSION)
 message(FATAL_ERROR "pass -DNFVS_PAYLOAD_DLL=<plug-in dll> -DNFVS_VERSION=X.Y.Z")
endif()
configure_file(installer.rc.in ${CMAKE_BINARY_DIR}/installer.rc @ONLY)
add_executable(NFVSInstaller WIN32 installer.cpp ${CMAKE_BINARY_DIR}/installer.rc)
target_compile_definitions(NFVSInstaller PRIVATE NFVS_VERSION="${NFVS_VERSION}" NFVS_WVERSION=L"${NFVS_VERSION}" UNICODE _UNICODE)
target_link_libraries(NFVSInstaller PRIVATE shell32 ole32 advapi32 user32)
target_link_options(NFVSInstaller PRIVATE /MANIFEST:NO)
